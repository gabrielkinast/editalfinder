-- DDL espelha public.noticia para manter mesmo contrato nas telas (ajuste índices se precisar).
-- Execute no SQL Editor do Supabase antes de usar a rota Pesquisas.

create table public.pesquisa (
  id uuid not null default gen_random_uuid (),
  titulo text not null,
  resumo text null,
  conteudo text null,
  link text not null,
  fonte text null,
  fonte_recurso text null,
  data_publicacao date null,
  prazo_envio date null,
  autor text null,
  pais text null,
  regiao text null,
  idioma text null,
  idioma_original text null,
  origem_portal text null,
  tipo_conteudo text null,
  content_type text not null default 'pesquisa'::text,
  imagem_url text null,
  area_cientifica text[] null,
  area_tecnologica text[] null,
  setor_estrategico text[] null,
  tags text[] null,
  perfil_ideal text[] null,
  publico_alvo text null,
  qualidade_dado integer null,
  validacao_status text null default 'incompleto'::text,
  codigo_oportunidade text null,
  numero_chamada text null,
  extras jsonb not null default '{}'::jsonb,
  ativo boolean not null default true,
  hash_deduplicacao text null,
  ultima_coleta timestamp with time zone null,
  criado_em timestamp with time zone not null default now(),
  atualizado_em timestamp with time zone not null default now(),
  constraint pesquisa_pkey primary key (id),
  constraint pesquisa_link_key unique (link)
) TABLESPACE pg_default;

create index IF not exists idx_pesquisa_fonte_recurso on public.pesquisa using btree (fonte_recurso) TABLESPACE pg_default;
create index IF not exists idx_pesquisa_data_publicacao on public.pesquisa using btree (data_publicacao desc) TABLESPACE pg_default;
create index IF not exists idx_pesquisa_prazo_envio on public.pesquisa using btree (prazo_envio) TABLESPACE pg_default;
create index IF not exists idx_pesquisa_validacao_status on public.pesquisa using btree (validacao_status) TABLESPACE pg_default;
create index IF not exists idx_pesquisa_tags_gin on public.pesquisa using gin (tags) TABLESPACE pg_default;
